Solution:
1. Collinear(a,c,d) => Parallel(a,c,c,d)
2. Perpendicular(a,c,b,d) & Parallel(a,c,c,d) => Perpendicular(b,d,c,d)
3. Midpoint(f,b,c) & Perpendicular(b,d,c,d) => Length_c_f - Length_d_f
4. Collinear(a,b,e) => Parallel(a,b,b,e)
5. Perpendicular(a,b,c,e) & Parallel(a,b,b,e) => Perpendicular(b,e,c,e)
6. Midpoint(f,b,c) & Perpendicular(b,e,c,e) => Length_c_f - Length_e_f
7. Length_c_f - Length_d_f & Length_c_f - Length_e_f => Length_d_f - Length_e_f
8. Length_d_f - Length_e_f => IsoscelesTriangle(d,e,f)
9. IsoscelesTriangle(d,e,f) & Length_d_f - Length_e_f & Perpendicular(d,e,f,g) => Length_d_g - Length_e_g
