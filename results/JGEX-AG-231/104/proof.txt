1. Collinear(a,c,p) => Parallel(a,c,a,p)
2. Perpendicular(a,c,m,p) & Parallel(a,c,a,p) => Perpendicular(a,p,m,p)
3. Perpendicular(a,p,m,p) => Angle_a_p_m - pi/2
4. Collinear(a,c,p) & Collinear(a,b,q) => Angle_b_a_p - Angle_c_a_q
5. Angle_c_a_n - Angle_c_a_q - Angle_n_a_q & Angle_b_a_m - Angle_b_a_p - Angle_m_a_p & Angle_a_p_m - pi/2 & Angle_b_a_p - Angle_c_a_q & Angle_b_a_m - Angle_c_a_n & Angle_a_m_p + Angle_a_p_m + Angle_m_a_p - pi => Angle_a_m_p + Angle_n_a_q - pi/2
6. Collinear(a,b,q) => Parallel(a,b,a,q)
7. Perpendicular(a,b,m,q) & Parallel(a,b,a,q) => Perpendicular(a,q,m,q)
8. Perpendicular(a,q,m,q) => Angle_a_q_m - pi/2
9. Angle_a_p_m - pi/2 & Angle_a_q_m - pi/2 => Angle_a_p_m + Angle_a_q_m - pi

12. Angle_a_m_p + Angle_n_a_q - pi/2 & Angle_a_m_p - Angle_a_q_p => Angle_a_q_p + Angle_n_a_q - pi/2
13. Angle_a_q_p + Angle_n_a_q - pi/2 => Perpendicular(a,n,p,q)
