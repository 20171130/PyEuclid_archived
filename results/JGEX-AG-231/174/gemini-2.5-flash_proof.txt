Auxiliary constructions:
k = construct_reflect(c, b, d), l = construct_reflect(c, a, e)
Solution:
1. Length_b_c - Length_b_k => IsoscelesTriangle(b,c,k)
2. Perpendicular(b,d,c,d) & Perpendicular(b,d,c,k) => Parallel(c,d,c,k)
3. Parallel(c,d,c,k) => Collinear(c,d,k)
4. IsoscelesTriangle(b,c,k) & Length_b_c - Length_b_k & Collinear(c,d,k) & Perpendicular(b,d,c,k) => Angle_c_b_d - Angle_d_b_k
5. Angle_c_b_d - Angle_d_b_k & Angle_a_b_c - 2*Angle_c_b_d & Angle_c_b_d - Angle_c_b_k + Angle_d_b_k => Angle_a_b_c - Angle_c_b_k
6. Angle_a_b_c - Angle_c_b_k => Collinear(a,b,k)
7. Collinear(a,b,k) => Parallel(a,b,a,k)
8. Length_a_c - Length_a_l => IsoscelesTriangle(a,c,l)
9. Perpendicular(a,e,c,e) & Perpendicular(a,e,c,l) => Parallel(c,e,c,l)
10. Parallel(c,e,c,l) => Collinear(c,e,l)
11. IsoscelesTriangle(a,c,l) & Length_a_c - Length_a_l & Collinear(c,e,l) & Perpendicular(a,e,c,l) => Angle_c_a_e - Angle_e_a_l
12. Angle_c_a_e - Angle_e_a_l & Angle_b_a_c - 2*Angle_c_a_e & Angle_c_a_e - Angle_c_a_l + Angle_e_a_l => Angle_b_a_c - Angle_c_a_l
13. Angle_b_a_c - Angle_c_a_l => Collinear(a,b,l)
14. Collinear(a,b,k) & Collinear(a,b,l) => Collinear(a,k,l)
15. Collinear(a,k,l) => Parallel(a,k,k,l)
16. Length_c_d - Length_d_k & Collinear(c,d,k) => Midpoint(d,c,k)
17. Length_c_e - Length_e_l & Collinear(c,e,l) => Midpoint(e,c,l)
18. Midpoint(d,c,k) & Midpoint(e,c,l) => Parallel(d,e,k,l)
19. Parallel(a,k,k,l) & Parallel(d,e,k,l) => Parallel(a,k,d,e)
20. Parallel(a,b,a,k) & Parallel(a,k,d,e) => Parallel(a,b,d,e)
