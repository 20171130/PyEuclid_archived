Auxiliary constructions:
h = construct_foot(g, c, d)
Solution:
1. Perpendicular(a,f,c,d) & Perpendicular(b,e,c,d) => Parallel(a,f,b,e)
2. Perpendicular(a,f,c,d) & Perpendicular(c,d,g,h) => Parallel(a,f,g,h)
3. Collinear(c,d,e) & Collinear(c,d,f) => Collinear(c,e,f)
4. Collinear(c,d,e) & Collinear(c,d,h) => Collinear(c,e,h)
5. Collinear(c,e,f) & Collinear(c,e,h) => Collinear(e,f,h)
6. Parallel(a,f,b,e) & Parallel(a,f,g,h) & Collinear(a,b,g) & Collinear(e,f,h) => Length_a_g/Length_f_h - Length_b_g/Length_e_h
7. Length_a_g - Length_b_g & Length_a_g/Length_f_h - Length_b_g/Length_e_h => Length_e_h - Length_f_h
8. Length_e_h - Length_f_h & Collinear(e,f,h) => Midpoint(h,e,f)
9. Collinear(c,d,e) => Parallel(c,d,c,e)
10. Perpendicular(c,d,g,h) & Parallel(c,d,c,e) => Perpendicular(c,e,g,h)
11. Collinear(c,e,f) => Parallel(c,e,e,f)
12. Perpendicular(c,e,g,h) & Parallel(c,e,e,f) => Perpendicular(e,f,g,h)
13. Midpoint(h,e,f) & Perpendicular(e,f,g,h) => Length_e_g - Length_f_g
