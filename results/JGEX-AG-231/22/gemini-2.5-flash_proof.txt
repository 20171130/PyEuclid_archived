2. Perpendicular(a,f,c,d) & Parallel(c,d,c,e) => Perpendicular(a,f,c,e)

4. Collinear(c,e,f) => Parallel(c,e,e,f)

6. Perpendicular(a,f,c,d) & Perpendicular(b,e,c,d) => Parallel(a,f,b,e)
7. Length_e_h - Length_f_h & Length_a_g - Length_b_g => Length_a_g/Length_b_g - Length_f_h/Length_e_h
