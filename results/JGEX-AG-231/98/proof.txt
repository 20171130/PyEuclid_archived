Solution:
1. Rectangle(a,b,c,d) => Parallel(a,d,b,c) & Length_a_c - Length_b_d & Parallelogram(a,b,c,d)
2. Midpoint(g,a,e) & Midpoint(f,d,e) => Parallel(a,d,f,g)
3. Parallel(a,d,b,c) & Parallel(a,d,f,g) => Parallel(b,c,f,g)
4. Parallel(b,c,f,g) => Trapezoid(b,c,f,g)
5. Parallelogram(a,b,c,d) & Collinear(a,c,e) & Collinear(b,d,e) => Length_a_e - Length_c_e & Length_b_e - Length_d_e
6. Length_a_c - Length_b_d & Length_a_e - Length_c_e & Length_b_d - Length_b_e - Length_d_e & Length_a_c - Length_a_e - Length_c_e & Length_b_e - Length_d_e => Length_b_e - Length_c_e
7. Length_a_c - Length_b_d & Length_b_e - Length_b_f + Length_e_f & Length_b_d - Length_b_f - Length_d_f & Length_a_g - Length_e_g & Length_c_e - Length_c_g + Length_e_g & Length_a_c - Length_a_g - Length_c_g & Length_d_f - Length_e_f & Length_b_e - Length_c_e => Length_b_f - Length_c_g
8. Trapezoid(b,c,f,g) & Parallel(b,c,f,g) & Length_b_f - Length_c_g => EquilateralTrapezoid(b,c,f,g)
9. EquilateralTrapezoid(b,c,f,g) & Parallel(b,c,f,g) => Length_b_g - Length_c_f
