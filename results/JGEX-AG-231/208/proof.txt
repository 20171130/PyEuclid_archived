Solution:
1. Parallel(a,b,c,d) => Trapezoid(a,b,d,c)
2. Trapezoid(a,b,d,c) & Parallel(a,b,c,d) & Length_a_d - Length_b_c => EquilateralTrapezoid(a,b,d,c)
3. EquilateralTrapezoid(a,b,d,c) & Parallel(a,b,c,d) => Length_a_c - Length_b_d & Angle_a_b_d - Angle_b_a_c
4. Collinear(c,d,e) => Parallel(c,d,d,e) & Parallel(c,d,c,e)
5. Parallel(a,b,c,d) & Parallel(c,d,d,e) => Parallel(a,b,d,e)
6. Parallel(a,b,d,e) => Angle_a_b_d + Angle_b_d_e - pi
7. Parallel(a,b,c,d) & Parallel(c,d,c,e) => Parallel(a,b,c,e)
8. Parallel(a,b,c,e) => Angle_a_c_e + Angle_b_a_c - pi
9. Angle_a_b_d - Angle_b_a_c & Angle_a_b_d + Angle_b_d_e - pi & Angle_a_c_e + Angle_b_a_c - pi => Angle_a_c_e - Angle_b_d_e
10. Length_a_c - Length_b_d & Length_c_e - Length_d_e & Angle_a_c_e - Angle_b_d_e => Congruent3(a,c,e,b,d,e)
11. Congruent3(a,c,e,b,d,e) => Length_a_e - Length_b_e
12. Length_a_e - Length_b_e => IsoscelesTriangle(a,b,e)
13. IsoscelesTriangle(a,b,e) & Length_a_e - Length_b_e & Collinear(a,b,f) & Midpoint(f,a,b) => Perpendicular(a,b,e,f)
