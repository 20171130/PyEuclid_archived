Solution:
1. Parallel(a,b,c,d) => Trapezoid(a,b,c,d)
2. Trapezoid(a,b,c,d) & Parallel(a,b,c,d) & Length_a_d - Length_b_c => EquilateralTrapezoid(a,b,c,d)
3. EquilateralTrapezoid(a,b,c,d) & Parallel(a,b,c,d) => Angle_a_b_c - Angle_b_a_d
4. Collinear(c,d,e) => Parallel(c,d,d,e) & Parallel(c,d,c,e)
5. Parallel(a,b,c,d) & Parallel(c,d,d,e) => Parallel(a,b,d,e)
6. Parallel(a,b,d,e) => Angle_a_d_e + Angle_b_a_d - pi
7. Parallel(a,b,c,d) & Parallel(c,d,c,e) => Parallel(a,b,c,e)
8. Parallel(a,b,c,e) => Angle_a_b_c + Angle_b_c_e - pi
9. Angle_a_b_c - Angle_b_a_d & Angle_a_d_e + Angle_b_a_d - pi & Angle_a_b_c + Angle_b_c_e - pi => Angle_a_d_e - Angle_b_c_e
10. -Length_a_d + Length_b_c & Length_c_e - Length_d_e & -Angle_a_d_e + Angle_b_c_e => Congruent3(a,d,e,b,c,e)
11. Congruent3(a,d,e,b,c,e) => Length_a_e - Length_b_e
12. Length_a_e - Length_b_e => IsoscelesTriangle(a,b,e)
13. IsoscelesTriangle(a,b,e) & Length_a_e - Length_b_e & Collinear(a,b,f) & Midpoint(f,a,b) => Perpendicular(a,b,e,f)
