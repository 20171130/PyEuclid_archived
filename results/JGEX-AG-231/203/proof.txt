2. Length_a_c - Length_a_d & Collinear(a,c,d) => Midpoint(a,c,d)
3. Midpoint(a,c,d) => Length_a_c - Length_c_d/2
4. Length_a_c - Length_b_c & Length_b_d - 2*Length_b_f & Length_a_c - Length_c_d/2 => Length_b_d/Length_c_d - Length_b_f/Length_b_c
5. Midpoint(e,a,c) => Length_a_c - 2*Length_c_e

7. Collinear(a,c,d) & Collinear(a,c,e) => Collinear(c,d,e)
8. Collinear(c,d,e) => Angle_b_c_d - Angle_b_c_e
9. Length_b_c/Length_c_e - Length_c_d/Length_b_c & Angle_b_c_d - Angle_b_c_e => Similar3(b,c,d,e,c,b)
10. Similar3(b,c,d,e,c,b) => Length_b_d/Length_b_e - Length_c_d/Length_b_c
