Solution:
1. Midpoint(e,b,c) & Midpoint(f,c,d) => Parallel(b,d,e,f)
2. Midpoint(h,a,b) & Midpoint(g,a,d) => Parallel(b,d,g,h)
3. Parallel(b,d,e,f) & Parallel(b,d,g,h) => Parallel(e,f,g,h)
4. Midpoint(h,a,b) & Midpoint(e,b,c) => Parallel(a,c,e,h)
5. Midpoint(g,a,d) & Midpoint(f,c,d) => Parallel(a,c,f,g)
6. Parallel(a,c,e,h) & Parallel(a,c,f,g) => Parallel(e,h,f,g)
7. Parallel(e,f,g,h) & Parallel(e,h,f,g) => Parallelogram(e,f,g,h)
8. Parallelogram(e,f,g,h) & Collinear(e,g,i) & Collinear(f,h,i) => Length_f_i - Length_h_i
9. Length_f_i - Length_h_i & Collinear(f,h,i) => Midpoint(i,f,h)
