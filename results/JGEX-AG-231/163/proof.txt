4. Parallel(a,b,a,g) & Parallel(a,b,c,h) => Parallel(a,g,c,h)

6. Collinear(a,d,f) => Parallel(a,d,a,f)
7. Collinear(b,c,i) => Parallel(b,c,c,i)
8. Parallel(a,d,b,c) & Parallel(b,c,c,i) => Parallel(a,d,c,i)
