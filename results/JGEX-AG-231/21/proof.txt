1. Midpoint(o,c,d) & Perpendicular(c,e,d,e) => Length_c_o - Length_e_o & Length_d_o - Length_e_o
2. -Length_c_o + Length_e_o & -Length_d_o + Length_e_o & Perpendicular(a,e,e,o) => Angle_a_e_c - Angle_c_d_e
3. Collinear(d,e,f) => Angle_a_f_d - Angle_a_f_e & Parallel(d,e,e,f)
4. Collinear(c,d,o) => Parallel(c,d,d,o)
5. Perpendicular(a,c,c,d) & Parallel(c,d,d,o) => Perpendicular(a,c,d,o)
6. Collinear(a,c,f) => Parallel(a,c,a,f)
7. Perpendicular(a,c,d,o) & Parallel(a,c,a,f) => Perpendicular(a,f,d,o)
8. Perpendicular(a,f,d,o) => Angle_a_f_d + Angle_f_d_o - pi/2
9. Collinear(d,e,f) & Collinear(c,d,o) => Angle_c_d_e - Angle_f_d_o
10. Angle_a_f_d - Angle_a_f_e & Angle_a_f_d + Angle_f_d_o - pi/2 & Angle_c_d_e - Angle_f_d_o => Angle_a_f_e + Angle_c_d_e - pi/2
11. Perpendicular(c,e,d,e) & Parallel(d,e,e,f) => Perpendicular(c,e,e,f)
12. Perpendicular(c,e,e,f) => Angle_c_e_f - pi/2
13. Angle_a_e_c + Angle_a_e_f - Angle_c_e_f & Angle_c_e_f - pi/2 => Angle_a_e_c + Angle_a_e_f - pi/2
14. Angle_a_e_c - Angle_c_d_e & Angle_a_f_e + Angle_c_d_e - pi/2 & Angle_a_e_c + Angle_a_e_f - pi/2 => Angle_a_e_f - Angle_a_f_e
15. Angle_a_e_f - Angle_a_f_e => Length_a_e - Length_a_f
