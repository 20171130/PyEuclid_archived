Solution:
1. Collinear(c,d,e) => Angle_b_c_d - Angle_b_c_e & Angle_a_d_c - Angle_a_d_e
2. Parallel(a,b,c,d) => Trapezoid(a,b,c,d)
3. Trapezoid(a,b,c,d) & Parallel(a,b,c,d) & -Length_a_d + Length_b_c => EquilateralTrapezoid(a,b,c,d)
4. EquilateralTrapezoid(a,b,c,d) & Parallel(a,b,c,d) => Angle_a_d_c - Angle_b_c_d
5. Angle_b_c_d - Angle_b_c_e & Angle_a_d_c - Angle_b_c_d & Angle_a_d_c - Angle_a_d_e => Angle_a_d_e - Angle_b_c_e
6. -Length_a_d + Length_b_c & Length_c_e - Length_d_e & -Angle_a_d_e + Angle_b_c_e => Congruent3(a,d,e,b,c,e)
7. Congruent3(a,d,e,b,c,e) => Length_a_e - Length_b_e
8. Length_a_e - Length_b_e => IsoscelesTriangle(a,b,e)
9. IsoscelesTriangle(a,b,e) & Length_a_e - Length_b_e & Collinear(a,b,f) & Midpoint(f,a,b) => Perpendicular(a,b,e,f)
