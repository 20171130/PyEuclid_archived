1. Collinear(a,c,e) => Parallel(a,c,a,e)
2. Midpoint(f,a,b) & Midpoint(d,b,c) => Parallel(a,c,d,f)
3. Parallel(a,c,a,e) & Parallel(a,c,d,f) => Parallel(a,e,d,f)
4. Collinear(a,b,f) => Parallel(a,b,a,f)
5. Midpoint(e,a,c) & Midpoint(d,b,c) => Parallel(a,b,d,e)
6. Parallel(a,b,a,f) & Parallel(a,b,d,e) => Parallel(a,f,d,e)
7. Parallel(a,e,d,f) & Parallel(a,f,d,e) => Parallelogram(a,e,d,f)
8. Parallelogram(a,e,d,f) => Angle_e_a_f + Angle_e_d_f*(-1)
9. Collinear(b,c,g) => Parallel(b,c,c,g) & Parallel(b,c,b,g)
10. Perpendicular(a,g,b,c) & Parallel(b,c,c,g) => Perpendicular(a,g,c,g)
11. Midpoint(e,a,c) & Perpendicular(a,g,c,g) => Length_a_e + Length_e_g*(-1)
12. Length_a_e + Length_e_g*(-1) => Length_a_e - Length_e_g
13. Perpendicular(a,g,b,c) & Parallel(b,c,b,g) => Perpendicular(a,g,b,g)
14. Midpoint(f,a,b) & Perpendicular(a,g,b,g) => Length_a_f + Length_f_g*(-1)
15. Length_a_f + Length_f_g*(-1) => Length_a_f - Length_f_g
16. Length_a_e - Length_e_g & -Length_a_f + Length_f_g => Kite(a,e,g,f)
17. Kite(a,e,g,f) & Length_a_e - Length_e_g & -Length_a_f + Length_f_g => Angle_e_a_f + Angle_e_g_f*(-1)
18. Angle_e_a_f + Angle_e_d_f*(-1) & Angle_e_a_f + Angle_e_g_f*(-1) => Angle_e_d_f + Angle_e_g_f*(-1)
19. Angle_e_d_f + Angle_e_g_f*(-1) => Angle_e_d_f - Angle_e_g_f
