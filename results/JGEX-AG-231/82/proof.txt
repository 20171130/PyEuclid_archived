2. Midpoint(g,a,b) & Midpoint(e,b,c) => Parallel(a,c,e,g)
3. Midpoint(e,b,c) & Midpoint(f,c,d) => Parallel(b,d,e,f)
4. Collinear(a,b,g) => Parallel(a,b,a,g)
5. Collinear(a,b,d) => Parallel(a,b,b,d)
6. Parallel(a,b,a,g) & Parallel(a,b,b,d) => Parallel(a,g,b,d)
7. Collinear(a,b,g) & Collinear(a,b,i) => Collinear(a,g,i)
8. Collinear(a,g,i) => Parallel(a,g,g,i)
9. Parallel(a,g,b,d) & Parallel(a,g,g,i) => Parallel(b,d,g,i)
10. Parallel(b,d,e,f) & Parallel(b,d,g,i) => Parallel(e,f,g,i)
11. Collinear(e,h,i) & Collinear(f,g,h) & Parallel(e,f,g,i) => Length_e_h/Length_f_h - Length_e_i/Length_f_g
12. Midpoint(h,f,g) => Length_f_g - 2*Length_f_h
13. Length_e_h/Length_f_h - Length_e_i/Length_f_g & Length_f_g - 2*Length_f_h => Length_e_h/Length_e_i - 1/2
14. Length_e_h/Length_e_i - 1/2 & Collinear(e,h,i) => Midpoint(h,e,i)
15. Midpoint(h,e,i) & Midpoint(h,f,g) => Parallel(e,g,f,i)
16. Parallel(a,c,e,g) & Parallel(e,g,f,i) => Parallel(a,c,f,i)
17. Collinear(a,d,i) & Collinear(c,d,f) & Parallel(a,c,f,i) => Length_a_d/Length_c_d - Length_a_i/Length_c_f
18. Midpoint(f,c,d) => Length_c_d - 2*Length_c_f
19. Length_a_d/Length_c_d - Length_a_i/Length_c_f & Length_c_d - 2*Length_c_f => -1/2 + Length_a_i/Length_a_d
20. -1/2 + Length_a_i/Length_a_d & Collinear(a,d,i) => Midpoint(i,a,d)
21. Midpoint(i,a,d) => Length_a_i - Length_d_i
