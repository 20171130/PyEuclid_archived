1. Square(a,b,d,e) => Length_a_b - Length_b_d & Angle_a_b_d - Angle_b_a_e & Angle_a_b_d + Angle_b_a_e - pi
2. Collinear(a,f,g) => Parallel(a,f,a,g)
3. Perpendicular(a,f,b,c) & Parallel(a,f,a,g) => Perpendicular(a,g,b,c)
4. Perpendicular(a,g,b,c) => Angle_a_b_c + Angle_b_a_g - pi/2 & Angle_a_g_b + Angle_c_b_g - pi/2
5. Angle_a_b_d - Angle_b_a_e & Angle_a_b_d + Angle_b_a_e - pi & Angle_a_b_c - Angle_a_b_d + Angle_c_b_d & Angle_a_b_c + Angle_b_a_g - pi/2 => Angle_b_a_g - Angle_c_b_d
6. Length_a_b - Length_b_d & Length_a_g - Length_b_c & Angle_b_a_g - Angle_c_b_d => Congruent3(a,b,g,b,d,c)
7. Congruent3(a,b,g,b,d,c) => Angle_a_g_b - Angle_b_c_d
8. Angle_a_g_b - Angle_b_c_d & Angle_b_c_g + Angle_b_g_c + Angle_c_b_g - pi & Angle_a_g_b + Angle_c_b_g - pi/2 & Angle_b_c_d - Angle_b_c_g + Angle_d_c_g => Angle_b_g_c + Angle_d_c_g - pi/2
9. Angle_b_g_c + Angle_d_c_g - pi/2 => Perpendicular(b,g,c,d)
