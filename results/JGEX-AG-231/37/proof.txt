1. Parallel(b,c,d,e) => Trapezoid(b,c,d,e) & Angle_b_e_d + Angle_c_b_e - pi
2. Length_a_b - Length_a_e & Length_a_b - Length_a_d => Length_a_d - Length_a_e
3. -Length_a_b + Length_a_e & -Length_a_d + Length_a_e => Angle_b_a_d + 2*Angle_b_e_d - 2*pi
4. Length_a_b - Length_a_c & Length_a_b - Length_a_d => Length_a_c - Length_a_d
5. -Length_a_b + Length_a_c & Length_a_c - Length_a_d => Angle_b_a_d - 2*Angle_b_c_d
6. Angle_b_a_d + 2*Angle_b_e_d - 2*pi & Angle_b_a_d - 2*Angle_b_c_d & Angle_b_e_d + Angle_c_b_e - pi => Angle_b_c_d - Angle_c_b_e
7. Trapezoid(b,c,d,e) & Parallel(b,c,d,e) & -Angle_b_c_d + Angle_c_b_e => EquilateralTrapezoid(b,c,d,e)
8. EquilateralTrapezoid(b,c,d,e) & Parallel(b,c,d,e) => Length_b_e - Length_c_d
