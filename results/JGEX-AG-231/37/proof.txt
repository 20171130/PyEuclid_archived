4. Length_a_b + Length_a_e*(-1) => Length_a_b - Length_a_e

7. Angle_b_c_d + Angle_c_b_e*(-1) => Angle_b_c_d - Angle_c_b_e
8. Trapezoid(b,c,d,e) & Parallel(b,c,d,e) & -Angle_b_c_d + Angle_c_b_e => EquilateralTrapezoid(b,c,d,e)
