2. Length_a_c - Length_b_c & Length_a_f - Length_b_f => Perpendicular(a,b,c,f)
3. Perpendicular(a,b,b,d) & Perpendicular(a,b,c,f) => Parallel(b,d,c,f)
4. Collinear(b,e,f) & Collinear(c,d,e) & Parallel(b,d,c,f) => Similar3(b,d,e,f,c,e)
5. Similar3(b,d,e,f,c,e) => Length_c_e/Length_e_f - Length_d_e/Length_b_e & Length_b_e/Length_e_f - Length_d_e/Length_c_e
