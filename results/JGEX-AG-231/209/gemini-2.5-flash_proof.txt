Auxiliary constructions:
m = construct_midpoint(a, b)
Solution:
1. Midpoint(m,a,b) & Perpendicular(a,c,b,c) => Length_b_m - Length_c_m
2. -Length_a_m + Length_b_m & Length_b_m - Length_c_m => Angle_a_b_c - Angle_a_m_c/2
3. Angle_a_b_c - Angle_c_b_d & Angle_a_b_c - Angle_a_m_c/2 & Angle_a_b_c - Angle_a_b_d + Angle_c_b_d => Angle_a_b_d - Angle_a_m_c
4. Angle_a_b_d - Angle_a_m_c & Collinear(a,b,m) => Parallel(b,d,c,m)
5. Midpoint(m,a,b) & Midpoint(e,a,d) => Similar3(a,b,d,a,m,e)
6. Similar3(a,b,d,a,m,e) => Angle_a_b_d - Angle_a_m_e
7. Angle_a_b_c - Angle_c_b_d & Angle_a_b_c - Angle_a_m_c/2 & Angle_a_b_c - Angle_a_b_d + Angle_c_b_d & Angle_a_b_d - Angle_a_m_e => Angle_a_m_c - Angle_a_m_e
8. Angle_a_m_c - Angle_a_m_e => Collinear(c,e,m)
9. Collinear(c,e,m) => Parallel(c,e,c,m)
10. Parallel(b,d,c,m) & Parallel(c,e,c,m) => Parallel(b,d,c,e)
