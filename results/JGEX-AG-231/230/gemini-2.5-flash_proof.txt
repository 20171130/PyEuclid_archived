3. Length_a_g - Length_b_e & Length_a_c - Length_b_c => Length_a_c/Length_a_g - Length_b_c/Length_b_e

5. Parallel(a,b,a,f) & Parallel(a,b,e,g) => Parallel(a,f,e,g)
