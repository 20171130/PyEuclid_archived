6. Parallel(a,b,d,h) & Parallel(a,b,e,f) => Parallel(d,h,e,f)

8. Collinear(a,c,f) & Collinear(a,c,i) => Collinear(c,f,i)

10. Collinear(b,c,e) => Parallel(b,c,c,e)
11. Midpoint(d,a,b) & Midpoint(f,a,c) => Parallel(b,c,d,f)

14. Midpoint(d,a,b) & Midpoint(e,b,c) => Parallel(a,c,d,e)
15. Parallel(a,c,c,f) & Parallel(a,c,d,e) => Parallel(c,f,d,e)
16. Parallel(c,e,d,f) & Parallel(c,f,d,e) => Parallelogram(c,e,d,f)
17. Parallelogram(c,e,d,f) => Angle_e_c_f - Angle_e_d_f
18. Collinear(e,h,i) & Collinear(b,d,h) => Angle_b_h_i - Angle_d_h_e
19. Length_e_g - Length_f_g & -Length_d_g + Length_e_g & Perpendicular(e,g,e,h) => Angle_e_d_f + Angle_f_e_h - pi

21. Angle_b_c_i - Angle_b_h_i => Concyclic(b,c,h,i)
