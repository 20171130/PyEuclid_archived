1. Midpoint(p,a,b) & Midpoint(q,b,c) => Parallel(a,c,p,q)
2. Midpoint(s,a,d) & Midpoint(r,c,d) => Parallel(a,c,r,s)
3. Parallel(a,c,p,q) & Parallel(a,c,r,s) => Parallel(p,q,r,s)
4. Midpoint(p,a,b) & Midpoint(s,a,d) => Parallel(b,d,p,s)
5. Midpoint(q,b,c) & Midpoint(r,c,d) => Parallel(b,d,q,r)
6. Parallel(b,d,p,s) & Parallel(b,d,q,r) => Parallel(p,s,q,r)
7. Parallel(p,q,r,s) & Parallel(p,s,q,r) => Parallelogram(p,q,r,s)
8. Parallelogram(p,q,r,s) & Collinear(o,p,r) & Collinear(o,q,s) => Length_o_p - Length_o_r
9. Length_o_p - Length_o_r => Midpoint(o,p,r)
10. Perpendicular(a,c,b,d) & Parallel(b,d,p,s) => Perpendicular(a,c,p,s)
11. Perpendicular(a,c,p,s) & Parallel(a,c,r,s) => Perpendicular(p,s,r,s)
12. Midpoint(o,p,r) & Perpendicular(p,s,r,s) => Length_o_r - Length_o_s
