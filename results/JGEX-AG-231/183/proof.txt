1. Midpoint(f,b,c) & Midpoint(e,c,d) => Similar3(b,c,d,f,c,e)

3. Midpoint(e,c,d) => Length_c_d + Length_d_e*(-2)
4. Midpoint(f,b,c) => Length_b_c + Length_c_f*(-2)
5. Length_c_d + Length_d_e*(-2) & Length_b_c + Length_c_f*(-2) => Length_b_c/Length_c_d + Length_c_f*(-1)/Length_d_e
6. Parallel(a,b,c,d) => Trapezoid(a,b,d,c)
7. Length_a_d + Length_b_c*(-1) => Length_a_d - Length_b_c
8. Trapezoid(a,b,d,c) & Parallel(a,b,c,d) & Length_a_d - Length_b_c => EquilateralTrapezoid(a,b,d,c)
9. EquilateralTrapezoid(a,b,d,c) & Parallel(a,b,c,d) => Length_a_c + Length_b_d*(-1)
10. Midpoint(h,a,d) & Midpoint(e,c,d) => Similar3(a,c,d,h,e,d)
11. Similar3(a,c,d,h,e,d) => Length_d_e/Length_c_d + Length_e_h*(-1)/Length_a_c

13. Length_e_f + Length_e_h*(-1) => Length_e_f - Length_e_h
