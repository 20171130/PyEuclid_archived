1. Midpoint(h,a,d) & Midpoint(e,c,d) => Similar3(a,c,d,h,e,d)
2. Similar3(a,c,d,h,e,d) => Length_a_c/Length_e_h - Length_c_d/Length_d_e
3. Midpoint(f,b,c) & Midpoint(e,c,d) => Similar3(b,c,d,f,c,e)
4. Similar3(b,c,d,f,c,e) => Length_b_c/Length_c_f - Length_b_d/Length_e_f
5. Parallel(a,b,c,d) => Trapezoid(a,b,c,d)
6. Trapezoid(a,b,c,d) & Parallel(a,b,c,d) & -Length_a_d + Length_b_c => EquilateralTrapezoid(a,b,c,d)
7. EquilateralTrapezoid(a,b,c,d) & Parallel(a,b,c,d) => Length_a_c - Length_b_d
8. Midpoint(e,c,d) => Length_c_d - 2*Length_d_e
9. Midpoint(f,b,c) => Length_b_c - 2*Length_c_f
10. Length_c_d - 2*Length_d_e & Length_b_c - 2*Length_c_f => Length_d_e/Length_c_f - Length_c_d/Length_b_c
11. Length_a_c/Length_e_h - Length_c_d/Length_d_e & Length_b_c/Length_c_f - Length_b_d/Length_e_f & Length_a_c - Length_b_d & Length_d_e/Length_c_f - Length_c_d/Length_b_c => Length_e_f - Length_e_h
