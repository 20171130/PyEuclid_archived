Solution:
1. Length_b_o + Length_c_o*(-1) => Length_b_o - Length_c_o
2. Length_a_o + Length_b_o*(-1) => Length_a_o - Length_b_o
3. Perpendicular(a,o,a,p) => Angle_o_a_p - pi/2
4. Length_b_o - Length_c_o & -Length_a_o + Length_b_o & Angle_o_a_p - pi/2 & Collinear(b,c,p) => Length_a_p/Length_c_p + Length_b_p*(-1)/Length_a_p
5. Length_a_p + Length_d_p*(-1) & Length_a_p/Length_c_p + Length_b_p*(-1)/Length_a_p => Length_b_p/Length_d_p + Length_d_p*(-1)/Length_c_p
6. Length_b_p/Length_d_p + Length_d_p*(-1)/Length_c_p => Length_b_p/Length_d_p - Length_d_p/Length_c_p
7. Collinear(b,c,p) => Angle_b_p_d + Angle_c_p_d*(-1)
8. Angle_b_p_d + Angle_c_p_d*(-1) => Angle_b_p_d - Angle_c_p_d
9. Length_b_p/Length_d_p - Length_d_p/Length_c_p & Angle_b_p_d - Angle_c_p_d => Similar3(b,d,p,d,c,p)
10. Similar3(b,d,p,d,c,p) => Angle_c_d_p + Angle_d_b_p*(-1)
11. Collinear(b,d,e) & Collinear(b,c,p) => Angle_c_b_e + Angle_d_b_p*(-1)
12. Length_b_o + Length_e_o*(-1) => Length_b_o - Length_e_o
13. Length_b_o + Length_c_o*(-1) & Length_c_o + Length_f_o*(-1) => Length_b_o + Length_f_o*(-1)
14. Length_b_o + Length_f_o*(-1) => Length_b_o - Length_f_o
15. Length_b_o - Length_c_o & Length_b_o - Length_e_o & Length_b_o - Length_f_o => Concyclic(b,c,e,f)
16. Concyclic(b,c,e,f) => Angle_c_b_e + Angle_c_f_e*(-1)
17. Angle_c_b_e + Angle_d_b_p*(-1) & Angle_c_b_e + Angle_c_f_e*(-1) => Angle_c_f_e + Angle_d_b_p*(-1)
18. Angle_c_d_p + Angle_d_b_p*(-1) & Angle_c_f_e + Angle_d_b_p*(-1) => Angle_c_d_p + Angle_c_f_e*(-1)
19. Angle_c_d_p + Angle_c_f_e*(-1) => Angle_c_d_p - Angle_c_f_e
20. Angle_c_d_p - Angle_c_f_e & Collinear(c,d,f) => Parallel(d,p,e,f)
