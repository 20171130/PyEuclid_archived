1. Collinear(c,d,e) => Parallel(c,d,c,e) & Angle_a_c_d - Angle_a_c_e
2. Perpendicular(a,b,c,d) & Parallel(c,d,c,e) => Perpendicular(a,b,c,e)
3. Perpendicular(a,b,c,e) => Angle_a_c_e + Angle_b_a_c - pi/2
4. Perpendicular(a,c,b,c) => Angle_a_c_b - pi/2
5. Angle_a_c_e + Angle_b_a_c - pi/2 & Angle_a_c_b - pi/2 & Angle_a_c_d - Angle_a_c_f + Angle_d_c_f & Angle_b_a_c - 2*Angle_c_a_e & Angle_a_c_b - Angle_a_c_f - Angle_b_c_f & Angle_b_c_f - Angle_d_c_f & Angle_a_c_d - Angle_a_c_e => Angle_a_c_f + Angle_c_a_e - pi/2
6. Angle_a_c_f + Angle_c_a_e - pi/2 => Perpendicular(a,e,c,f)
7. Collinear(a,b,f) => Parallel(a,b,a,f)
8. Perpendicular(a,b,c,e) & Parallel(a,b,a,f) => Perpendicular(a,f,c,e)
9. Perpendicular(a,e,c,f) & Perpendicular(a,f,c,e) => Orthocenter(a,c,e,f)
10. Orthocenter(a,c,e,f) => Perpendicular(a,c,e,f)
11. Perpendicular(a,c,b,c) & Perpendicular(a,c,e,f) => Parallel(b,c,e,f)
