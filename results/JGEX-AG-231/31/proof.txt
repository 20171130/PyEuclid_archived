3. Midpoint(g,b,c) & Perpendicular(b,d,c,d) => Length_c_g - Length_d_g

6. Perpendicular(a,b,c,d) & Parallel(c,d,d,e) => Perpendicular(a,b,d,e)
7. Perpendicular(a,b,d,e) & Parallel(a,b,a,d) => Perpendicular(a,d,d,e)
8. Midpoint(f,a,e) & Perpendicular(a,d,d,e) => Length_d_f - Length_e_f

10. Collinear(c,d,e) & Collinear(b,c,g) => Angle_b_c_e - Angle_d_c_g
11. Collinear(a,e,f) => Parallel(a,e,e,f)
12. Perpendicular(a,e,b,c) & Parallel(a,e,e,f) => Perpendicular(b,c,e,f)
13. Perpendicular(b,c,e,f) => Angle_b_c_e + Angle_c_e_f - pi/2
14. Angle_c_e_f - Angle_d_e_f & Angle_b_c_e - Angle_d_c_g & Angle_b_c_e + Angle_c_e_f - pi/2 => Angle_d_c_g + Angle_d_e_f - pi/2
