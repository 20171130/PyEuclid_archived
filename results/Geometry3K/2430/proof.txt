1. Parallelogram(A,C,B,D) => Area_A_C_B_D - Length_A_C*Length_B_C*sin(Angle_A_C_B) & Length_A_C - Length_B_D & Parallel(A,D,B,C) & Angle_A_C_B + Angle_C_B_D - pi
2. Collinear(A,D,E) => Parallel(A,D,D,E) & Parallel(A,E,D,E) & Angle_A_D_B - Angle_B_D_E & Angle_B_A_D - Angle_B_A_E
3. Parallel(A,D,B,C) & Parallel(A,D,D,E) => Parallel(B,C,D,E)
4. Parallel(B,C,D,E) => Angle_B_D_E + Angle_C_B_D - pi
5. Angle_B_D_E + Angle_C_B_D - pi & Angle_B_D_E - pi/3 & Angle_A_C_B + Angle_C_B_D - pi => Angle_A_C_B - pi/3
6. Area_A_C_B_D - Length_A_C*Length_B_C*sin(Angle_A_C_B) & Length_A_C - Length_B_D & Angle_A_C_B - pi/3 & Length_B_C - 14 => Area_A_C_B_D - 7*sqrt(3)*Length_B_D
7. Perpendicular(A,E,B,E) & Parallel(A,E,D,E) => Perpendicular(B,E,D,E)
8. Perpendicular(B,E,D,E) => Angle_B_E_D - pi/2
9. Angle_A_D_B - Angle_B_D_E & Angle_B_A_D - Angle_B_A_E & Angle_B_D_E - pi/3 & Angle_A_B_D + Angle_A_D_B + Angle_B_A_D - pi => Angle_A_B_D + Angle_B_A_E - 2*pi/3
10. Perpendicular(A,E,B,E) => Angle_A_E_B - pi/2
11. Angle_A_E_B - pi/2 & Angle_A_B_E + Angle_A_E_B + Angle_B_A_E - pi => Angle_A_B_E + Angle_B_A_E - pi/2
12. Angle_A_B_D - Angle_A_B_E - Angle_D_B_E & Angle_A_B_D + Angle_B_A_E - 2*pi/3 & Angle_A_B_E + Angle_B_A_E - pi/2 => Angle_D_B_E - pi/6
13. sin(Angle_D_B_E)/Length_D_E - sin(Angle_B_E_D)/Length_B_D & Angle_B_E_D - pi/2 & Length_D_E - 7 & Angle_D_B_E - pi/6 => Length_B_D - 14
14. Area_A_C_B_D - 7*sqrt(3)*Length_B_D & Length_B_D - 14 => Area_A_C_B_D - 98*sqrt(3)
