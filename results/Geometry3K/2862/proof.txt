1. Collinear(C,D,N) => Angle_D_C_F - Angle_F_C_N
2. Collinear(B,D,E) => Parallel(B,D,B,E)
3. Parallel(A,F,B,E) & Parallel(B,D,B,E) => Parallel(A,F,B,D)
4. Collinear(A,C,F) => Parallel(A,F,C,F)
5. Parallel(A,F,B,D) & Parallel(A,F,C,F) => Parallel(B,D,C,F)
6. Parallel(B,D,C,F) => Angle_B_D_C + Angle_D_C_F - pi
7. Collinear(C,D,N) & Collinear(B,D,E) => Angle_B_D_C - Angle_E_D_N
8. Angle_D_C_F - Angle_F_C_N & Angle_B_D_C + Angle_D_C_F - pi & Angle_B_D_C - Angle_E_D_N => Angle_E_D_N + Angle_F_C_N - pi
