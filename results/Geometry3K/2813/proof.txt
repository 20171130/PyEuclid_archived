Solution:
1. Length_V_X - Variable_radius_V & Length_V_W - Variable_radius_V => Length_V_W - Length_V_X
2. Length_V_X - Variable_radius_V & Length_V_Y - Variable_radius_V => Length_V_X - Length_V_Y
3. -Length_V_W + Length_V_X & Length_V_X - Length_V_Y => Angle_W_V_Y + 2*Angle_W_X_Y - 2*pi
4. Length_V_W - Variable_radius_V & Length_V_Z - Variable_radius_V => Length_V_W - Length_V_Z
5. Length_V_Y - Variable_radius_V & Length_V_Z - Variable_radius_V => Length_V_Y - Length_V_Z
6. -Length_V_W + Length_V_Z & -Length_V_Y + Length_V_Z => Angle_W_V_Y - 2*Angle_W_Z_Y
7. Angle_W_V_Y + 2*Angle_W_X_Y - 2*pi & Angle_W_V_Y - 2*Angle_W_Z_Y & Angle_W_Z_Y - pi/3 => Angle_W_X_Y - 2*pi/3
