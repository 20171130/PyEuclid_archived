1. Rhombus(A,B,C,D) => Perpendicular(A,C,B,D)
2. Collinear(B,D,E) => Parallel(B,D,B,E)
3. Perpendicular(A,C,B,D) & Parallel(B,D,B,E) => Perpendicular(A,C,B,E)
4. Collinear(A,C,E) => Parallel(A,C,A,E)
5. Perpendicular(A,C,B,E) & Parallel(A,C,A,E) => Perpendicular(A,E,B,E)
6. Perpendicular(A,E,B,E) => Angle_A_E_B - pi/2
