1. Collinear(A,C,D) => Parallel(A,D,C,D) & Parallel(A,C,C,D)
2. Perpendicular(A,B,C,D) & Parallel(A,D,C,D) => Perpendicular(A,B,A,D)
3. Perpendicular(A,B,A,D) => Angle_B_A_D - pi/2

6. Perpendicular(A,B,C,D) & Parallel(A,C,C,D) => Perpendicular(A,B,A,C)

10. Similar3(A,B,C,A,D,B) => Length_A_B/Length_A_D - Length_A_C/Length_A_B
