2. Collinear(B,D,E) => Parallel(B,D,D,E) & Parallel(B,E,D,E)
3. Perpendicular(A,E,D,E) & Parallel(B,D,D,E) => Perpendicular(A,E,B,D)
4. IsoscelesTriangle(A,B,D) & Length_A_B - Length_A_D & Perpendicular(A,E,B,D) => Length_B_E - Length_D_E
5. Length_B_E - Length_D_E & Length_D_E - 5 => Length_B_E - 5
6. Perpendicular(A,E,D,E) & Parallel(B,E,D,E) => Perpendicular(A,E,B,E)
7. Collinear(A,C,E) => Parallel(A,E,C,E)
8. Perpendicular(A,E,B,E) & Parallel(A,E,C,E) => Perpendicular(B,E,C,E)
9. Perpendicular(B,E,C,E) => Angle_B_E_C - pi/2
10. Length_B_C**2 - Length_B_E**2 + 2*Length_B_E*Length_C_E*cos(Angle_B_E_C) - Length_C_E**2 & Length_C_E - 12 & Length_B_C - Length_C_D & Length_B_E - 5 & Angle_B_E_C - pi/2 => Length_C_D - 13
11. Length_B_C - Length_C_D & Length_C_D - 13 => Length_B_C - 13
