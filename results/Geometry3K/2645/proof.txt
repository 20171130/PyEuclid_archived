1. Length_D_E - 5 & Length_C_E - 12 => Length_C_E - 12*Length_D_E/5
2. Length_A_B - Length_A_D => IsoscelesTriangle(A,B,D)
3. Collinear(B,D,E) => Parallel(B,D,D,E)
4. Perpendicular(A,E,D,E) & Parallel(B,D,D,E) => Perpendicular(A,E,B,D)
5. IsoscelesTriangle(A,B,D) & Length_A_B - Length_A_D & Collinear(B,D,E) & Perpendicular(A,E,B,D) => Midpoint(E,B,D)
6. Midpoint(E,B,D) => Length_B_D - 2*Length_D_E
7. Length_C_E - 12*Length_D_E/5 & Length_B_D - 2*Length_D_E & Length_C_E - 12 => Length_B_D - 10
8. Collinear(A,C,E) => Parallel(A,E,C,E)
9. Perpendicular(A,E,B,D) & Parallel(A,E,C,E) => Perpendicular(B,D,C,E)
10. Perpendicular(B,D,C,E) & Collinear(B,D,E) => Area_B_C_D - Length_B_D*Length_C_E/2
11. Length_B_D - 10 & Area_B_C_D - Length_B_D*Length_C_E/2 & Length_C_E - 12 => Area_B_C_D - 60

13. Length_B_C - Length_C_D & Length_C_D - 13 => Length_B_C - 13
